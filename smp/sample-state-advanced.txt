#type state

#   the top-level pseudo-states
Start     [ state: initial ]
Split     [ state: fork ]
Merge     [ state: join ]
Route     [ state: junction ]
End       [ state: final ]
Abort     [ state: terminate ]

#   the composite state with its pseudo-states
Editing   [ entry: "lock()", exit: "unlock()" ]
Begin     [ state: initial, parent: Editing ]
Resume    [ state: history, parent: Editing ]
Leave     [ state: exit-point, parent: Editing ]
Draft     [ parent: Editing ]
Proof     [ parent: Editing, exit: "save()" ]
Done      [ state: final, parent: Editing ]

#   the regular states
Suspended
Payment   [ do: "charge()" ]
Shipping  [ do: "pack()" ]
Review

#   the transitions (the completion transition "Editing --> Split" fires
#   once "Done" is reached, and the two "ok" transitions conflict)
Start     --> Editing
Begin     --> Draft
Draft     --("submit")--> Proof
Draft     --("discard")--> Leave
Leave     --> End
Proof     --("approve")--> Done
Proof     --("reject")--> Draft
Editing   --("suspend")--> Suspended
Suspended --("resume")--> Resume
Suspended --("cancel / cleanup()")--> Abort
Editing   --> Split
Split     --> Payment
Split     --> Shipping
Payment   --("paid")--> Merge
Shipping  --("packed")--> Merge
Merge     --> Route
Route     --("[express] / notify()")--> End
Route     --("[else]")--> Review
Review    --("ok")--> End
Review    --("ok")--> Editing
